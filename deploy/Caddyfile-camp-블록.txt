# camp.kodekorea.kr — 창의디자인캠프 원클릭 설치 파일 배포
#
# /etc/caddy/Caddyfile 맨 아래에 이 블록만 추가한다. 다른 블록은 손대지 않는다.
# 이 서버에는 16개 도메인이 함께 돌고 있다.
#
# 설계 의도: 주소에 접속하면 **아무것도 누르지 않고 바로 받아진다.**
# 페이지도 단추도 없다. 초등학교 5학년이 헤맬 자리를 만들지 않는다.
# 받은 파일을 두 번 누르면 스스로 풀리고 설치까지 이어진다.
#
# 주소는 ASCII 로 두고 받는 파일 이름만 한글로 준다.
# 한글이 들어간 주소는 브라우저·셸·메신저를 거치며 깨진다(실측으로 겪었다).
#
# header 값에 따옴표를 쓰지 않는 이유: Caddy 가 그것을 치환(replace) 인자로
# 해석해서 헤더가 아예 안 붙는다. 이것도 실측으로 잡았다.
#
# 이 파일 안에는 API 열쇠가 들어 있다. 검색엔진에 잡히면 그 자체가 사고다.
# 색인·아카이브를 헤더로 막고, 캠프가 끝나면 이 블록을 지운다.

camp.kodekorea.kr {
	header {
		X-Robots-Tag "noindex, nofollow, noarchive, nosnippet"
		Referrer-Policy "no-referrer"
		-Server
	}

	root * /var/www/camp

	# 루트로 오면 곧 다운로드다. 잘못 알려진 주소들도 같은 곳으로 보낸다.
	@get path / /index.html /setup /setup.exe /install /camp2026-setup.exe
	handle @get {
		header Content-Disposition "attachment; filename=camp2026-setup.exe; filename*=UTF-8''%EC%B0%BD%EC%9D%98%EB%94%94%EC%9E%90%EC%9D%B8%EC%BA%A0%ED%94%84%20%EC%84%A4%EC%B9%98.exe"
		header Cache-Control "no-store"
		rewrite * /camp2026-setup.exe
		file_server
	}

	# 그 밖의 경로는 아무것도 보여주지 않는다 (폴더 목록 노출 방지)
	handle {
		respond 404
	}
}
